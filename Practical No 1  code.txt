Practical No: 1
Write menu based programs a) Write a program to store roll numbers of student in array who
attended a training program in random order. Write a function for searching whether a particular
student attended a training program or not, using Linear search

input:-
#include <iostream>
using namespace std;

// Function for Linear Search
int linearSearch(int arr[], int n, int key)
{
    for (int i = 0; i < n; i++)
    {
        if (arr[i] == key)
        {
            return i;
        }
    }
    return -1;
}

int main()
{
    int arr[100], n, key;

    cout << "Enter number of students: ";
    cin >> n;

    if (n <= 0 || n > 100)
    {
        cout << "Invalid number of students.";
        return 0;
    }

    cout << "Enter roll numbers of students: ";
    for (int i = 0; i < n; i++)
    {
        cin >> arr[i];
    }

    cout << "Enter roll number to search: ";
    cin >> key;

    int result = linearSearch(arr, n, key);

    if (result != -1)
    {
        cout << "Student attended the training program.";
    }
    else
    {
        cout << "Student did not attend the training program.";
    }

    return 0;
}

#include <stdio.h>

void binarySearch(int arr[], int n, int key)
{
    int low = 0, high = n - 1;
    int mid, found = 0;

    while (low <= high)
    {
        mid = low + (high - low) / 2;

        if (arr[mid] == key)
        {
            printf("Student with roll number %d attended the training program.\n", key);
            found = 1;
            break;
        }
        else if (arr[mid] < key)
        {
            low = mid + 1;
        }
        else
        {
            high = mid - 1;
        }
    }

    if (!found)
    {
        printf("Student with roll number %d did NOT attend the training program.\n", key);
    }
}

int main()
{
    int n, i, key;
    int arr[50];

    printf("Enter number of students: ");
    scanf("%d", &n);

    if (n <= 0 || n > 50)
    {
        printf("Invalid number of students.\n");
        return 0;
    }

    printf("Enter roll numbers of students in sorted order:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);

        if (i > 0 && arr[i] < arr[i - 1])
        {
            printf("Roll numbers must be in sorted order.\n");
            return 0;
        }
    }

    printf("Enter roll number to search: ");
    scanf("%d", &key);

    binarySearch(arr, n, key);

    return 0;
}
